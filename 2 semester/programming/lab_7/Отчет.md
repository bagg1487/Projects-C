## Оценка 3. Знакомство с pthread

### 1. Создание потока
Программа создает поток с помощью `pthread_create()` с атрибутами по умолчанию. Оба потока выводят по 5 строк текста.

### 2. Ожидание потока
Модифицированная программа, где родительский поток ждет завершения дочернего с помощью `pthread_join()` перед выводом своего текста.

### 3. Параметры потока
Основной поток создает 4 потока, каждый получает свой текстовый параметр и выводит разные последовательности строк.

### 4. Завершение потока без ожидания
Добавлен `sleep()` в потоки. Через 2 секунды основной поток отменяет дочерние с помощью `pthread_cancel()`.

### 5. Обработка завершения потока
Доработка с использованием `pthread_cleanup_push()` для вывода сообщения перед завершением потока.

```c
#include <stdio.h>
#include <pthread.h>
#include <unistd.h>

void clean(void *arg) {
    char* message = (char*)arg;
    printf("Поток %s завершен\n", message);
}

void *function(void *arg) {
    char* message = (char*)arg;
    pthread_cleanup_push(clean, message);
    
    for (int i = 1; i < 4; i++) {
        printf("Поток %d %s%d\n", i, message, i);
        sleep(1);
    }
    
    pthread_cleanup_pop(0);
    return NULL;
}

int main() {
    pthread_t thread[4];
    char* message[] = {"num_1.", "num_2.", "num_3.", "num_4."};

    for (int i = 0; i < 4; i++)
        pthread_create(&thread[i], NULL, function, message[i]);

    sleep(2);

    for (int i = 0; i < 4; i++)
        pthread_cancel(thread[i]);
        
    for (int i = 0; i < 4; i++)
        pthread_join(thread[i], NULL);
}
```

## Реализация Sleepsort

Алгоритм сортировки O(N), где каждый элемент массива "спит" пропорционально своему значению перед выводом.

```c
#include <stdio.h>
#include <pthread.h>
#include <unistd.h>
#include <stdlib.h>

void *sleepsort(void *arg) {
    int thr = *(int*)arg;
    sleep(thr);
    printf("%d ", thr);
    return NULL;
}

int main() {
    int arr[] = {1, 4, 3, 2, 9, 10, 8};
    int n = sizeof(arr) / sizeof(int);
    pthread_t thread[n];
    
    for (int i = 0; i < n; i++) {
        pthread_create(&thread[i], NULL, sleepsort, &arr[i]);
    }
    
    for (int i = 0; i < n; i++) {
        pthread_join(thread[i], NULL);
    }
    
    printf("\n");
}
```

## Оценка 4. Перемножение матриц

### Синхронизированный вывод
Модифицированная программа с взаимной блокировкой через mutex для поочередного вывода строк.

```c
#include <stdio.h>
#include <pthread.h>
#include <unistd.h>

pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
int turn = 0;

void* child_thread(void* arg) {
    char* message = (char*)arg;
    
    for (int i = 1; i < 6; i++) {
        pthread_mutex_lock(&mutex);
        while (turn != 1) {
            pthread_mutex_unlock(&mutex);
            usleep(100);
            pthread_mutex_lock(&mutex);
        }
        
        printf("Дочерний поток %s: %d\n", message, i);
        turn = 0;
        pthread_mutex_unlock(&mutex);
    }
    return NULL;
}

int main() {
    pthread_t thread;
    char* message = "test";
    
    pthread_create(&thread, NULL, child_thread, message);

    for (int i = 1; i < 6; i++) {
        pthread_mutex_lock(&mutex);
        while (turn != 0) {
            pthread_mutex_unlock(&mutex);
            usleep(100);
            pthread_mutex_lock(&mutex);
        }
        
        printf("Главный поток: %d\n", i);
        turn = 1;
        pthread_mutex_unlock(&mutex);
    }

    pthread_join(thread, NULL);
    pthread_mutex_destroy(&mutex);
}
```

### Перемножение квадратных матриц

Программа для параллельного перемножения матриц с замером времени выполнения.

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <sys/time.h>

int THREADS, N;
int **A, **B, **C;

void* multiplication(void* arg) {
    int thread_id = *(int*)arg;
    int rows_per_thread = N / THREADS;
    int start = thread_id * rows_per_thread;
    int end = (thread_id == THREADS - 1) ? N : start + rows_per_thread;
    
    for (int i = start; i < end; i++) {
        for (int j = 0; j < N; j++) {
            C[i][j] = 0;
            for (int k = 0; k < N; k++) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }
    return NULL;
}

int main(int argc, char* argv[]) {
    if (argc != 3) {
        printf("Usage: %s <size> <threads>\n", argv[0]);
        return 1;
    }

    N = atoi(argv[1]);
    THREADS = atoi(argv[2]);

    // Выделение памяти и инициализация матриц
    A = malloc(N * sizeof(int*));
    B = malloc(N * sizeof(int*));
    C = malloc(N * sizeof(int*));
    
    for (int i = 0; i < N; i++) {
        A[i] = malloc(N * sizeof(int));
        B[i] = malloc(N * sizeof(int));
        C[i] = malloc(N * sizeof(int));
        
        for (int j = 0; j < N; j++) {
            A[i][j] = 1;
            B[i][j] = 1;
        }
    }

    pthread_t threads[THREADS];
    int thread_ids[THREADS];
    struct timeval start, end;

    gettimeofday(&start, NULL);

    // Создание потоков
    for (int i = 0; i < THREADS; i++) {
        thread_ids[i] = i;
        pthread_create(&threads[i], NULL, multiplication, &thread_ids[i]);
    }

    // Ожидание завершения потоков
    for (int i = 0; i < THREADS; i++) {
        pthread_join(threads[i], NULL);
    }

    gettimeofday(&end, NULL);
    double time = (end.tv_sec - start.tv_sec) + 
                 (end.tv_usec - start.tv_usec) / 1e6;

    // Вывод результатов для маленьких матриц
    if (N < 5) {
        printf("Matrix A:\n");
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                printf("%d ", A[i][j]);
            }
            printf("\n");
        }
        
        // Аналогично для B и C
    }

    printf("Execution time: %.6f seconds\n", time);

    // Освобождение памяти
    for (int i = 0; i < N; i++) {
        free(A[i]); free(B[i]); free(C[i]);
    }
    free(A); free(B); free(C);
    
    return 0;
}
```
